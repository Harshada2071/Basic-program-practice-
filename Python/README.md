# All Operators in One Python Program

a = 10
b = 5

# Arithmetic Operators 
print("Addition:", a + b)
print("Subtraction:", a - b)
print("Multiplication:", a * b)
print("Division:", a / b)
print("Floor Division:", a // b)
print("Modulus:", a % b)
print("Exponent:", a ** b)
# Assignment Operators 
x = 10
print("Initial Value:", x)

x += 5
print("x += 5 :", x)

x -= 3
print("x -= 3 :", x)

x *= 2
print("x *= 2 :", x)

x /= 4
print("x /= 4 :", x)

x //= 2
print("x //= 2 :", x)

x %= 3
print("x %= 3 :", x)

x **= 2
print("x **= 2 :", x)

# Comparison Operators 
print("a == b :", a == b)
print("a != b :", a != b)
print("a > b :", a > b)
print("a < b :", a < b)
print("a >= b :", a >= b)
print("a <= b :", a <= b)

# Logical Operators 
print("a > 5 and b < 10 :", a > 5 and b < 10)
print("a < 5 or b < 10 :", a < 5 or b < 10)
print("not(a > b) :", not(a > b))

# Bitwise Operators 
print("a & b :", a & b)
print("a | b :", a | b)
print("a ^ b :", a ^ b)
